#include <iostream>
#include <vector>
#include <algorithm> // Required for std::sort

int main() {
    std::vector<int> numbers = {64, 34, 25, 12, 22, 11, 90};

    // Sorts the range from beginning to end in ascending order
    std::sort(numbers.begin(), numbers.end());

    std::cout << "STL Sorted vector: ";
    for (int num : numbers) {
        std::cout << num << " ";
    }
    
    return 0;
}
